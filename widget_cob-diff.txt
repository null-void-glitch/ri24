        function submitCalcToChat() {
            Object.keys(CALC_INPUT_LABELS).forEach(function (key) {
                const field = document.querySelector('#ri24-baufi-calc-input-rows .ri24-baufi-calc-input-field[data-key="' + key + '"]');

                    calcState.input[key] = field.value.trim();

            document.getElementById('ri24-baufi-calc-status').style.display = 'none';
            markCalcLive();
            // Ergebnis-Felder sofort auf "wird berechnet" zurücksetzen, statt das alte (jetzt nicht mehr zu den neuen Eingaben passende) Ergebnis stehen zu lassen, bis die Antwort (calc_result per SSE) eintrifft.
            Object.keys(CALC_RESULT_LABELS).forEach(function (key) {
                calcState.result[key] = '…';

            renderCalcRows('ri24-baufi-calc-result-rows', CALC_RESULT_LABELS, calcState.result);
            const parts = Object.keys(CALC_INPUT_LABELS)
                .filter(function (key) {
                    return calcState.input[key];
                })
                .map(function (key) {
                    return CALC_INPUT_LABELS[key] + ': ' + calcState.input[key];
                });

        document.getElementById('ri24-baufi-calc-recalc-btn').addEventListener('click', submitCalcToChat);
        renderCalcInputRows();
        renderCalcRows('ri24-baufi-calc-result-rows', CALC_RESULT_LABELS, calcState.result);

        // Höhe des Rechner-Panels im eingeklappten Zustand einmalig messen — wird
        // unten als feste Chat-Höhe verwendet, damit beide Elemente gleich hoch
        // wirken, auch nachdem der Rechner nach der ersten Berechnung aufklappt
        // (Chat-Höhe bleibt dabei unverändert).
        const calcPanelHeight = document.getElementById('ri24-baufi-calc-panel').getBoundingClientRect().height;
        const windowPanelHeight = document.getElementById('ri24-baufi-chat-calc-stack').getBoundingClientRect().height;
        const calcPanelCollapsedHeight = (calcPanelHeight > 100 ? calcPanelHeight : (windowPanelHeight > 100 ? windowPanelHeight : 600)) - 32;

        // ── DeepChat konfigurieren, sobald das Custom-Element definiert ist ───────
        // Das Modul-Bundle (type="module") lädt deferred. Würden wir chat.connect
        // etc. sofort setzen, wäre <deep-chat> noch nicht "upgraded" und DeepChat
        // startet im Demo-Modus ("can't answer that question"). whenDefined
        // stellt sicher, dass die Konfiguration greift.
        Promise.all([customElements.whenDefined('deep-chat'), advisorPromise]).then(function (results) {
            const advisor = results[1];

            // ── Verbindung: eigener Stream-Handler statt connect.url + requestInterceptor —
            // nur so lassen sich neben dem Chat-Text auch die zusätzlichen SSE-Felder
            // (calc_input/calc_result) aus chat.php auswerten.
            chat.connect = {
                stream: true,
                handler: function (body, signals) {
                    body.conversation_id = conversationId;
                    body.berater_url = beraterUrl;
                    body.berater_id = parseInt('4');

                    const controller = new AbortController();
                    signals.stopClicked.listener = function () {
                        controller.abort();
                    };

                    fetch(ENDPOINT, {
                        method: 'POST',
                        headers: {'Content-Type': 'application/json'},
                        body: JSON.stringify(body),
                        signal: controller.signal
                    }).then(function (res) {
                        if (!res.ok || !res.body) {
                            handleAssistantTurnWithoutCalc();
                            signals.onResponse({error: 'Der Assistent ist momentan nicht erreichbar.'});
                            signals.onClose();
                            return;
                        }
                        signals.onOpen();
                        const reader = res.body.getReader();
                        const decoder = new TextDecoder();
                        let buffer = '';

                        function pump() {
                            return reader.read().then(function (chunk) {
                                if (chunk.done) {
                                    handleAssistantTurnWithoutCalc();
                                    signals.onClose();
                                    return;
                                }
                                buffer += decoder.decode(chunk.value, {stream: true});
                                const lines = buffer.split('\n');
                                buffer = lines.pop();
                                lines.forEach(function (line) {
                                    if (line.indexOf('data:') !== 0) {
                                        return;
                                    }
                                    const raw = line.slice(5).trim();
                                    if (raw === '' || raw === '[DONE]') {
                                        return;
                                    }
                                    let payload;
                                    try {
                                        payload = JSON.parse(raw);
                                    } catch (e) {
                                        return;
                                    }
                                    if (payload.text) {
                                        signals.onResponse({text: payload.text});
                                    } else if (payload.calc_input) {
                                        updateCalcInput(payload.calc_input);
                                    } else if (payload.calc_result) {
                                        updateCalcResult(payload.calc_result);
                                    }
                                });
                                return pump();
                            });
                        }

                        return pump();
                    }).catch(function (err) {
                        if (err && err.name === 'AbortError') {
                            handleAssistantTurnWithoutCalc();
                            signals.onClose();
                            return;
                        }
                        handleAssistantTurnWithoutCalc();
                        signals.onResponse({error: 'Verbindung zum Assistenten fehlgeschlagen.'});
                        signals.onClose();
                    });
                }
            };

            // ── Erscheinungsbild ─────────────────────────────────────────────────────
            chat.style =
                'border-radius:0 0 14px 14px;border:1px solid #e2e5ea;border-top:none;width:100%;max-width:100%;height:' +
                calcPanelCollapsedHeight + 'px;font-family:system-ui,-apple-system,Segoe UI,Roboto,sans-serif;';
            // DeepChat positioniert den Sendebutton per festem inset-block-end (kein
            // echtes vertikales Zentrieren) und vererbt input-button {margin-bottom}
            // auch an absolut positionierte Buttons — beides zusammen lässt ihn optisch
            // zu hoch sitzen. Bottom:50%+translateY(50%) zentriert ihn exakt unabhängig
            // von Button-/Zeilenhöhe; margin: 0 entfernt den zusätzlichen Versatz.
            const submitCenterFix = document.createElement('style');
            submitCenterFix.textContent =
                // .ri24-starter-Divs (width:100% + border) sind 1px breiter als ihr
                // Container — reicht auf Linux/GTK mit klassischen (nicht-Overlay)
                // Scrollbalken, damit #messages dauerhaft einen grauen horizontalen
                // Scrollbalken anzeigt, der optisch wie eine falsch positionierte
                // Trennlinie mitten im Chat wirkt.
                '* { overflow-x: hidden !important; } ' +
                // Der Sendebutton hängt (Position inside-end) absolut positioniert außerhalb
                // seines eigenen 0px breiten Containers (.input-button-container) — die
                // obige globale Regel würde ihn sonst unsichtbar clippen.
                '.input-button-container { overflow: visible !important; } ' +
                '.inside-end, .inside-start, .outside-end, .outside-start { ' +
                'inset-block-end: 50% !important; transform: translateY(50%) !important; margin: 0 !important; } ' +
                // .input-button ist standardmäßig display: block — das Icon-SVG darin
                // bleibt dadurch oben, statt vertikal zu zentrieren.
                '.input-button { display: flex !important; align-items: center !important; justify-content: center !important; } ' +
                // Während des Wartens (loading-button) bzw. während die Antwort streamt
                // (Stop-Button, erkennbar am #stop-icon-Kind) zeigt DeepChat den grünen
                // Kreis mit grauem Quadrat/Punkt — hier durch die schon vorhandene,
                // aber unsichtbar positionierte "drei springende Punkte"-Animation
                // (.loading-submit-button) ersetzt, ohne grünen Hintergrund.
                '.input-button.loading-button, .input-button:has(#stop-icon) { background-color: transparent !important; } ' +
                '.loading-submit-button { inset-block-end: 0 !important; } ' +
                '#stop-icon { position: relative !important; inset-inline-start: calc(-9990px + 0.275em) !important; ' +
                'inset-block-end: 0 !important; width: 0.22em !important; height: 0.22em !important; border-radius: 5px !important; ' +
                'background-color: #848484 !important; box-shadow: 9990px 0 0 0 #848484, calc(9990px + 0.44em) 0 0 0 #848484, ' +
                'calc(9990px + 0.8em) 0 0 0 #848484 !important; animation: loading-submit-button 1.5s infinite linear !important; }';
            chat.shadowRoot.appendChild(submitCenterFix);
            chat.textInput = {
                placeholder: {text: 'Deine Frage zur Baufinanzierung …'},
                styles: {text: {color: '#111'}}
            };
            chat.submitButtonStyles = {
                submit: {
                    container: {
                        default: {
                            backgroundColor: '#BDE04F',
                            color: '#333333',
                            width: '2rem',
                            height: '2rem',
                            borderRadius: '50%',
                            display: "flex",
                            justifyContent: "center",
                            alignItems: "center"
                        }
                    },
                    svg: {
                        styles: {
                            default: {
                                strokeWidth: '2px',
                                filter: 'none',
                                width: '45%',
                                height: '45%'
                            }
                        }
                    }
                }
            };
            chat.messageStyles = {
                default: {
                    ai: {bubble: {backgroundColor: '#f1f4f2', color: '#111'}},
                    user: {bubble: {backgroundColor: '#BDE04F', color: '#333'}}
                }
            };
            chat.speechToText = {
                webSpeech: {language: 'de-DE'},
                displayInterimResults: false,
                submitAfterSilence: 2500,
                commands: {stop: 'stopp', submit: 'absenden'},
                textColor: {final: '#333333'},
                button: {
                    active: {
                        container: {
                            default: {backgroundColor: '#BDE04F'}
                        }
                    }
                }
            };

            // ── Klickbare Gesprächseinstiege in der Intro-Nachricht ──────────────────
            chat.htmlClassUtilities = {
                'ri24-starter': {
                    events: {
                        click: function (event) {
                            chat.submitUserMessage({text: event.target.textContent.trim()});
                        }
                    },
                    styles: {
                        default: {
                            display: 'block', width: '100%', textAlign: 'left', cursor: 'pointer',
                            margin: '6px 0', padding: '10px 12px', borderRadius: '10px',
                            border: '1px solid #d7ddd9', backgroundColor: '#fff', color: '#333',
                            fontSize: '0.9rem'
                        },
                        hover: {backgroundColor: '#f1f4f2'}
                    }
                }
            };

            const greeting = advisor && advisor.name
                ? 'Ich bin der Baufinanzierungs-Assistent von <strong>' + advisor.name + '</strong>' +
                (advisor.title ? ' (' + advisor.title + ')' : '') + '.'
                : 'Ich bin dein Baufinanzierungs-Assistent.';

            chat.introMessage = {
                html:
                    '<div style="line-height:1.5;">' +
                    '<strong>👋 Willkommen!</strong><br>' + greeting + ' ' +
                    'Ich helfe dir, deine Finanzierung durchzurechnen und zu optimieren — ' +
                    'unabhängig von einem konkreten Exposé.' +
                    '<div style="margin-top:10px;">' +
                    '<div class="ri24-starter">Ich möchte eine Immobilienfinanzierung berechnen 🧮</div>' +
                    '<div class="ri24-starter">Wie kann ich meine Finanzierung optimieren? 📈</div>' +
                    '<div class="ri24-starter">Was ist eine Anschlussfinanzierung? ❓</div>' +
                    '<div class="ri24-starter">Ich möchte einen Beratungstermin buchen 📅</div>' +
                    '</div>' +
                    '</div>'
            };

        });
    })();
</script>
